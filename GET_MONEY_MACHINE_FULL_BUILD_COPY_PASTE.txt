def station_state(s):
    return {"id": s.id, "name": s.name, "active": s.active, "paused": s.paused, "emergency_stopped": s.emergency_stopped, "status": "EMERGENCY STOPPED" if s.emergency_stopped else "PAUSED" if s.paused else "RUNNING" if s.active else "STOPPED", "last_activity": s.last_activity}

def balance_values(db):
    completed = db.scalar(select(func.coalesce(func.sum(LedgerEntry.amount),0)).where(LedgerEntry.status=='COMPLETED')) or Decimal('0')
    reserved = db.scalar(select(func.coalesce(func.sum(Withdrawal.amount),0)).where(Withdrawal.status.in_(['PENDING','PROCESSING']))) or Decimal('0')
    return completed, reserved, completed-reserved

    if not s: raise HTTPException(404,"Station not found")

async def withdrawal(data:WithdrawalIn, _:Owner=Depends(current_owner), db:Session=Depends(get_db)):
    if db.scalar(select(Withdrawal).where(Withdrawal.idempotency_key==data.idempotency_key)): raise HTTPException(409,'Duplicate idempotency key')
    dest=db.get(Destination,data.destination_id)
    if not dest or not dest.enabled: raise HTTPException(404,'Destination unavailable')
    if dest.type.upper() != 'PAYPAL': raise HTTPException(400,'Only PayPal destinations are enabled')
    completed,reserved,available=balance_values(db)
    if data.amount > available: raise HTTPException(400,'Insufficient available ledger balance')
    w=Withdrawal(amount=data.amount,destination_id=dest.id,idempotency_key=data.idempotency_key,status='PENDING'); db.add(w); db.commit(); db.refresh(w)
    audit(db,'withdrawal.requested',f'id={w.id};amount={data.amount};provider=paypal')
    try:
        result=await paypal_service.create_payout(data.amount, dest.external_id, data.idempotency_key)
        w.status='PROCESSING'; w.provider_reference=result['provider_reference']; db.commit(); db.refresh(w)
    except Exception as e:
        w.status='FAILED'; w.error_message=str(getattr(e,'detail',e)); db.commit(); db.refresh(w)
    return w

async def reconcile(w, db):
    if not (w.provider_reference and w.status=='PROCESSING' and paypal_service.configured()): return w
    body=await paypal_service.get_payout(w.provider_reference)
    batch_status=body.get('batch_header',{}).get('batch_status')
    statuses={i.get('transaction_status') for i in body.get('items',[]) if i.get('transaction_status')}
    if 'FAILED' in statuses or batch_status == 'DENIED': w.status='FAILED'
    elif 'RETURNED' in statuses or 'REVERSED' in statuses: w.status='REVERSED'
    elif batch_status == 'SUCCESS' or 'SUCCESS' in statuses:
        w.status='COMPLETED'
        if not db.scalar(select(LedgerEntry).where(LedgerEntry.kind=='TRANSFER_DEBIT', LedgerEntry.provider_reference==w.provider_reference)):
            db.add(LedgerEntry(kind='TRANSFER_DEBIT', amount=-w.amount, status='COMPLETED', description='PayPal-confirmed transfer debit', provider_reference=w.provider_reference))
    db.commit(); db.refresh(w); return w

from fastapi import HTTPException
import httpx
from .config import settings


def base_url() -> str:
    return "https://api-m.paypal.com" if settings.paypal_env.lower() == "live" else "https://api-m.sandbox.paypal.com"


def configured() -> bool:
    return bool(settings.paypal_client_id and settings.paypal_client_secret and settings.paypal_webhook_id)


def setup_message() -> str:
    return "PAYPAL SETUP REQUIRED"

async def access_token() -> str:
    if not settings.paypal_client_id or not settings.paypal_client_secret:
        raise HTTPException(503, setup_message())
    async with httpx.AsyncClient(timeout=30) as client:
        r = await client.post(
            f"{base_url()}/v1/oauth2/token",
            auth=(settings.paypal_client_id, settings.paypal_client_secret),
            data={"grant_type": "client_credentials"},
            headers={"Accept": "application/json"},
        )
    if r.status_code >= 400:
        raise HTTPException(502, f"PayPal authentication failed ({r.status_code})")
    body = r.json()
    token = body.get("access_token")
    if not token:
        raise HTTPException(502, "PayPal authentication returned no access token")
    return token

async def create_payout(amount, recipient_email: str, reference: str):
    token = await access_token()
    batch_id = f"GMM-{reference}"
    payload = {
        "sender_batch_header": {
            "sender_batch_id": batch_id,
            "email_subject": "GET MONEY MACHINE payout",
            "email_message": "A payout has been initiated.",
        },
        "items": [{
            "recipient_type": "EMAIL",
            "amount": {"value": f"{amount:.2f}", "currency": "USD"},
            "receiver": recipient_email,
            "note": reference,
            "sender_item_id": reference,
        }],
    }
    async with httpx.AsyncClient(timeout=30) as client:
        r = await client.post(
            f"{base_url()}/v1/payments/payouts",
            json=payload,
            headers={
                "Authorization": f"Bearer {token}",
                "Content-Type": "application/json",
                "Accept": "application/json",
                "PayPal-Request-Id": batch_id,
            },
        )
    body = r.json() if r.content else {}
    if r.status_code >= 400:
        raise HTTPException(502, f"PayPal payout failed ({r.status_code}): {body.get('message') or body.get('name') or 'provider error'}")
    header = body.get("batch_header", {})
    return {"provider": "paypal", "status": header.get("batch_status", "PENDING"), "provider_reference": header.get("payout_batch_id")}

async def get_payout(payout_batch_id: str):
    token = await access_token()
    async with httpx.AsyncClient(timeout=30) as client:
        r = await client.get(
            f"{base_url()}/v1/payments/payouts/{payout_batch_id}",
            headers={"Authorization": f"Bearer {token}", "Accept": "application/json"},
        )
    body = r.json() if r.content else {}
    if r.status_code >= 400:
        raise HTTPException(502, f"PayPal payout lookup failed ({r.status_code})")
    return body

async def verify_webhook(headers: dict, event: dict):
    required = ["paypal-auth-algo", "paypal-cert-url", "paypal-transmission-id", "paypal-transmission-sig", "paypal-transmission-time"]
    if not settings.paypal_webhook_id or any(not headers.get(k) for k in required):
        raise HTTPException(400, "Invalid PayPal webhook headers")
    token = await access_token()
    payload = {
        "auth_algo": headers["paypal-auth-algo"],
        "cert_url": headers["paypal-cert-url"],
        "transmission_id": headers["paypal-transmission-id"],
        "transmission_sig": headers["paypal-transmission-sig"],
        "transmission_time": headers["paypal-transmission-time"],
        "webhook_id": settings.paypal_webhook_id,
        "webhook_event": event,
    }
    async with httpx.AsyncClient(timeout=30) as client:
        r = await client.post(
            f"{base_url()}/v1/notifications/verify-webhook-signature",
            json=payload,
            headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json", "Accept": "application/json"},
        )
    body = r.json() if r.content else {}
    if r.status_code >= 400 or body.get("verification_status") != "SUCCESS":
        raise HTTPException(400, "Invalid PayPal webhook signature")
    return True

from datetime import datetime, timedelta, timezone
import jwt
from argon2 import PasswordHasher
from argon2.exceptions import VerifyMismatchError
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from sqlalchemy.orm import Session
from .config import settings
from .db import get_db
from .models import Owner

ph = PasswordHasher()
bearer = HTTPBearer()

def hash_password(password): return ph.hash(password)
def verify_password(hash_value, password):
    try: return ph.verify(hash_value, password)
    except VerifyMismatchError: return False

def make_token(owner_id):
    return jwt.encode({"sub": str(owner_id), "exp": datetime.now(timezone.utc)+timedelta(hours=12)}, settings.jwt_secret, algorithm="HS256")

def current_owner(creds: HTTPAuthorizationCredentials = Depends(bearer), db: Session = Depends(get_db)):
    try: payload = jwt.decode(creds.credentials, settings.jwt_secret, algorithms=["HS256"])
    except jwt.PyJWTError: raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid session")
    owner = db.get(Owner, int(payload["sub"]))
    if not owner: raise HTTPException(status_code=401, detail="Invalid session")
    return owner

# GET MONEY MACHINE — Full Source Build

This repository contains the complete current source package for the GET MONEY MACHINE application: React mobile-first frontend, FastAPI backend, persistent SQL ledger, authentication, 15 stations, audit logging, scanner framework, PayPal payout integration, PayPal webhook verification, Render deployment configuration, Docker Compose, and automated tests.

## Important behavior
- Internal ledger is the source of truth; PayPal is the transfer provider.
- The seed creates exactly one $15,000 opening ledger entry.
- Discovery does not become available money without verification/settlement.
- No fabricated opportunities or fake payout confirmations are generated.
- Transfer requests require available internal ledger balance and PayPal provider processing.
- PayPal webhook signatures are verified server-side before webhook state changes.
- Provider success is required before a transfer is marked completed.
- PayPal secrets never belong in the frontend.

## Run locally
1. Copy `backend/.env.example` to `backend/.env` and set a strong JWT secret plus owner credentials.
2. `docker compose up --build`.
3. API: `http://localhost:8000/health`.
4. Frontend: `http://localhost:5173`.

## Render
`render.yaml` defines the API and Postgres resources. Set all `sync: false` variables in Render. After deployment, use the actual Render HTTPS service URL plus `/paypal/webhook` as the PayPal webhook endpoint.
